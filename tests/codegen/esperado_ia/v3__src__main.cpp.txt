if(texto.size()>4000){
returnstd::string("Mensagem longa demais (4000 bytes).");
}
intficha=localizar(texto);
if(ficha>=0){
returnresponder_ficha(ficha,texto);
}
std::stringseparado=regex_replace(minusculas(texto),std::string(" e "),std::string("\n"));
autopartes=string_split(separado,10);
if(partes.size()==2){
intprimeira=localizar(partes[0]);
intsegunda=localizar(partes[1]);
if(primeira>=0&&segunda>=0&&primeira!=segunda){
returnresponder_ficha(primeira,partes[0])+std::string("\n\n")+responder_ficha(segunda,partes[1]);

}
returnresponder_ficha(-1,texto);
}

std::stringtema_pesquisa(std::stringpergunta){

}

if(regex_match(texto,std::string("[a-z ?!.-]+"))==0){
returnstd::string("");
}
texto=regex_replace(texto,std::string(" +"),std::string(" "));
texto=regex_replace(texto,std::string("^ +|[ ?!.]+$"),std::string(""));
if(regex_search(texto,std::string("\\b(eu|meu|minha|meus|minhas|nosso|nossa|senha|segredo|token|chave|cpf|endereco|telefone|email|paciente|cliente|confidencial|interno|privado|diagnostico|tenho|sinto|estou)\\b"))){
returnstd::string("");
}
if(regex_search(texto,std::string("\\b(ghostlang|fantasmas|ghost|crie|execute|rode|apague|instale|envie|salve)\\b"))||regex_search(texto,std::string("(teoria|numeros) (dos )?g\\b"))){
returnstd::string("");
}
texto=regex_replace(texto,std::string("^(por favor )?(o que e|o que sao|quem e|quem foi|qual e|quais sao|como funciona|como funcionam|explique|descreva|fale sobre|pesquise sobre|pesquise|procure sobre|procure|busque sobre|busque) +"),std::string(""));
texto=regex_replace(texto,std::string("^(o|a|os|as|um|uma) +"),std::string(""));
if(regex_search(texto,std::string("\\b(voce|isso|aquilo|ele|ela|entendi|obrigado|ajude|pode)\\b"))){
returnstd::string("");
}
if(texto.size()<3||texto.size()>160){
returnstd::string("");
}
returntexto;
}

intcarregado=0;
intneural=0;
intassunto=-1;
intprogressivo=1;
intatraso=text_stream_delay();
intinternet=1;
std::stringultima_pergunta=std::string("");
std::stringultima_resposta=std::string("");
std::cout<<ghost_show(std::string("IA V3.4 | pesquisa automatica e memoria local com fontes."))<<std::endl;
std::cout<<ghost_show(std::string("/assistente | /neural | /fontes | /limpar | /sair"))<<std::endl;
std::cout<<ghost_show(std::string("/stream [on|off] | /velocidade 0..100 (ms por caractere)"))<<std::endl;

std::stringlinha=std::string("");
while(1){
ghost_print("Voce> ");

if(input_closed()){
return;
}
if(linha==std::string("/sair")){
return;
}
intcomando=0;
if(linha==std::string("/neural")){

if(file_exists(checkpoint)){

modelo.load(checkpoint);

neural=1;

}

comando=1;
neural=0;
std::cout<<ghost_show(std::string("Modo assistente: base revisada, memoria adquirida e pesquisa automatica quando falta conhecimento."))<<std::endl;
}

comando=1;
assunto=-1;
ultima_pergunta=std::string("");
ultima_resposta=std::string("");
std::cout<<ghost_show(std::string("Contexto da conversa limpo."))<<std::endl;
}

std::cout<<ghost_show(std::string("Streaming ativado. Use /stream off para desativar."))<<std::endl;

std::cout<<ghost_show(std::string("Streaming desativado. Use /stream on para ativar."))<<std::endl;

std::cout<<ghost_show(std::string("Streaming ativado."))<<std::endl;

if(linha==std::string("/stream off")){

progressivo=0;
std::cout<<ghost_show(std::string("Streaming desativado: resposta completa de uma vez."))<<std::endl;
}

comando=1;

std::cout<<ghost_show(std::string("Internet desativada. A base revisada e a memoria adquirida continuam disponiveis. /internet on reativa a pesquisa automatica."))<<std::endl;
}
}
if(linha==std::string("/internet on")){
comando=1;
internet=1;

}
if(linha==std::string("/internet off")){
comando=1;
internet=0;

}

if(regex_search(linha,std::string("^/velocidade"))){

}

exibir_resposta(resposta,progressivo,atraso);
}
if(tratada==0){

assunto=ficha;

std::stringprompt=std::string("Usuário: ")+linha+std::string("\nAssistente:");

resposta=modelo.stream(prompt,320,atraso,std::string("[Geracao neural] "));

}

else{

exibir_resposta(resposta,progressivo,atraso);
}
ultima_pergunta=linha;
ultima_resposta=resposta;
}
}
}

voidghost_main(GhostList<std::string>args){
std::cout<<ghost_show(device_info())<<std::endl;
GhostTextModelmodelo=text_model(64,32,192,32,4096,42);
if(args.size()==0){
conversa(std::string("checkpoints/v3/ghost_g_v31.bin"));
return;
}
std::stringcomando=args[0];
if(comando==std::string("consultar")&&args.size()==2){
std::cout<<ghost_show(consultar(args[1]))<<std::endl;
return;
}
if(comando==std::string("hardware")){
modelo.info();
return;
}
if(comando==std::string("autoteste")){
modelo.self_test();
return;
}
inttreino_solicitado=comando==std::string("treinar")||comando==std::string("continuar");
if(treino_solicitado){
inttamanho=6;
std::stringorigem=std::string("");
if(comando==std::string("continuar")){
tamanho=7;
}
if(args.size()!=tamanho){
text_error(std::string("Argumentos invalidos para treinamento."));
}
intepocas=text_int(args[3],1,200);
doubletaxa=text_decimal(args[4],0.000001,0.02);
std::stringdestino=args[tamanho-1];
intexiste=file_exists(destino)||file_exists(destino+std::string(".ultimo"))||file_exists(destino+std::string(".metricas.csv"))||file_exists(destino+std::string(".treino.txt"));
if(existe){
text_error(std::string("Destino ja existe. Use um novo nome para preservar o modelo."));
}
if(comando==std::string("continuar")){
origem=args[5];
}
modelo.prepare(args[1],args[2],origem);
writefile(destino+std::string(".treino.txt"),std::string("Treinamento local V3\nData: ")+format_date(now(),std::string("%Y-%m-%d %H:%M:%S %z"))+std::string("\nDispositivo: ")+device_info()+std::string("\nTreino: ")+args[1]+std::string("\nValidacao: ")+args[2]+std::string("\n"));
doublemelhor=modelo.validate();
std::cout<<ghost_show(std::string("CE inicial validacao: ")+melhor+std::string(" | perplexidade: ")+text_perplexity(melhor))<<std::endl;
modelo.save(destino);
intpaciencia=8;
intsem_melhora=0;
intepoca=0;
intcontinuar_treino=1;
std::stringmetricas=std::string("epoca,ce_treino_online,ce_validacao,perplexidade,acuracia\n");
while(continuar_treino){
doubleerro_treino=modelo.epoch(taxa);
doubleerro_validacao=modelo.validate();
doubleperplexidade=text_perplexity(erro_validacao);
doubleacerto=modelo.accuracy();
inttotal_epocas=modelo.epochs();
modelo.save(destino+std::string(".ultimo"));
if(erro_validacao<melhor){
melhor=erro_validacao;
sem_melhora=0;
modelo.save(destino);
}
else{
sem_melhora=sem_melhora+1;
}
std::cout<<ghost_show(std::string("Epoca ")+total_epocas+std::string(" | CE treino ")+erro_treino+std::string(" | CE validacao ")+erro_validacao+std::string(" | PPL ")+perplexidade+std::string(" | acuracia ")+acerto)<<std::endl;

writefile(destino+std::string(".metricas.csv"),metricas);
epoca=epoca+1;
continuar_treino=epoca<epocas&&sem_melhora<paciencia;
}
std::cout<<ghost_show(std::string("Melhor checkpoint: ")+destino+std::string(" | CE validacao ")+melhor)<<std::endl;

return;
}
intavaliar_solicitado=comando==std::string("avaliar")&&args.size()==3;
if(avaliar_solicitado){
modelo.load(args[2]);
doubleerro=modelo.evaluate(args[1]);

}
intgerar_solicitado=comando==std::string("gerar")&&args.size()==4;

if(chat_solicitado){
std::stringarquivo=std::string("checkpoints/v3/ghost_g_v31.bin");
if(args.size()==2){

conversa(arquivo);

}
intinfo_solicitado=comando==std::string("info")&&args.size()==2;
if(info_solicitado){
