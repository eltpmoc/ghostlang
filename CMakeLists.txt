cmake_minimum_required(VERSION 3.18)
project(GhostLang)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -fopenmp")

# Inclui os cabeçalhos da pasta include
include_directories(include)

if(WIN32)
    set(OS_LIBS ws2_32 wininet)
    # Garante que os executáveis do Ghost (ghost.exe, etc) sejam gerados sem depender de DLLs externas
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -static -static-libgcc -static-libstdc++")
else()
    set(OS_LIBS "")
endif()

# CUDA opcional: AUTO funciona tambem em maquinas sem toolkit/driver NVIDIA.
set(GHOST_CUDA "AUTO" CACHE STRING "CUDA: AUTO, ON ou OFF")
set_property(CACHE GHOST_CUDA PROPERTY STRINGS AUTO ON OFF)
if(NOT GHOST_CUDA MATCHES "^(AUTO|ON|OFF)$")
    message(FATAL_ERROR "GHOST_CUDA deve ser AUTO, ON ou OFF")
endif()
set(GHOST_RUNTIME_CUDA 0)
if(NOT GHOST_CUDA STREQUAL "OFF")
    if(NOT CMAKE_CUDA_COMPILER AND EXISTS "/usr/local/cuda/bin/nvcc")
        set(CMAKE_CUDA_COMPILER "/usr/local/cuda/bin/nvcc" CACHE FILEPATH "CUDA compiler")
    endif()
    include(CheckLanguage)
    check_language(CUDA)
    if(CMAKE_CUDA_COMPILER)
        if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
            set(CMAKE_CUDA_ARCHITECTURES "61;75;86;89")
        endif()
        enable_language(CUDA)
        find_package(CUDAToolkit REQUIRED)
        set(GHOST_RUNTIME_CUDA 1)
    elseif(GHOST_CUDA STREQUAL "ON")
        message(FATAL_ERROR "CUDA solicitado, mas nvcc nao foi encontrado")
    endif()
endif()
add_library(ghost_runtime STATIC src/GhostNumberG.cpp src/GhostDevice.cpp src/GhostTextModel.cpp src/GhostWeb.cpp src/GhostKnowledge.cpp src/GhostTemporal.cpp src/GhostTransformer.cpp src/GhostTransformerOps.cpp src/GhostTransformerG.cpp src/GhostMatrixG.cpp src/GhostQuantum.cpp src/GhostQML.cpp src/GhostHybrid.cpp src/GhostQTape.cpp src/sqlite3.c src/GhostDB.cpp src/GhostGUI.cpp src/GhostAutograd.cpp src/GhostDataset.cpp src/GhostDensityMatrix.cpp src/GhostNDTensor.cpp src/GhostTransformerFallback.cpp src/GhostQHardware.cpp src/GhostBLAS.cpp)
target_include_directories(ghost_runtime PUBLIC "${CMAKE_SOURCE_DIR}/include")
target_compile_definitions(ghost_runtime PUBLIC GHOST_NATIVE_RUNTIME=1)
# dlopen do backend CUDA do simulador quantico (Linux)
target_link_libraries(ghost_runtime PUBLIC ${CMAKE_DL_LIBS})
target_compile_options(ghost_runtime PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3>)
# OpenBLAS no dgemm dos tensores (src/GhostBLAS.cpp). Liga sozinho quando a biblioteca
# existe (gerada por scripts/build-openblas.ps1); -DGHOST_USE_BLAS=OFF desliga.
set(GHOST_OPENBLAS_LIB "${CMAKE_SOURCE_DIR}/third_party/openblas/lib/libopenblas.a" CACHE FILEPATH "libopenblas.a estatica")
if(EXISTS "${GHOST_OPENBLAS_LIB}")
    option(GHOST_USE_BLAS "Usa o OpenBLAS no dgemm dos tensores" ON)
else()
    option(GHOST_USE_BLAS "Usa o OpenBLAS no dgemm dos tensores" OFF)
endif()
if(GHOST_USE_BLAS)
    if(NOT EXISTS "${GHOST_OPENBLAS_LIB}")
        message(FATAL_ERROR "GHOST_USE_BLAS=ON, mas ${GHOST_OPENBLAS_LIB} nao existe (rode scripts/build-openblas.ps1)")
    endif()
    set_source_files_properties(src/GhostBLAS.cpp PROPERTIES COMPILE_DEFINITIONS GHOST_HAS_BLAS=1)
    add_custom_command(TARGET ghost_runtime POST_BUILD
        COMMAND ${CMAKE_COMMAND} -DRUNTIME=$<TARGET_FILE:ghost_runtime> -DBLAS=${GHOST_OPENBLAS_LIB} -DAR=${CMAKE_AR}
                -P ${CMAKE_SOURCE_DIR}/cmake/merge_blas.cmake
        COMMENT "Embutindo o OpenBLAS na libghost_runtime.a")
    message(STATUS "GhostLang: OpenBLAS ligado (${GHOST_OPENBLAS_LIB})")
else()
    message(STATUS "GhostLang: sem OpenBLAS; dgemm nativo")
endif()
set(GHOST_CUDART "")
if(GHOST_RUNTIME_CUDA)
    target_sources(ghost_runtime PRIVATE src/GhostCuda.cu src/GhostTextCuda.cu src/GhostTransformerCuda.cu)
    target_compile_definitions(ghost_runtime PUBLIC GHOST_HAS_CUDA=1)
    set_target_properties(ghost_runtime PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_link_libraries(ghost_runtime PUBLIC CUDA::cudart_static ${CMAKE_DL_LIBS})
    get_target_property(GHOST_CUDART CUDA::cudart_static IMPORTED_LOCATION)
endif()
message(STATUS "GhostLang runtime CUDA: ${GHOST_RUNTIME_CUDA}")
